// js/app.js - Main application script for the English course

document.addEventListener('DOMContentLoaded', () => {
   console.log("App.js: DOM fully loaded and parsed.");
   initializeCurrentLesson();
});

/**
* Determines the current lesson from the page and initializes its activities.
*/
function initializeCurrentLesson() {
   const lessonMeta = document.querySelector('meta[name="lesson-id"]');
   const lessonId = lessonMeta ? lessonMeta.content : null; // e.g., "week1"

   if (!lessonId) {
       console.log("App.js: Not on a lesson page or lesson ID meta tag not found.");
       return; // Exit if not a lesson page
   }

   console.log(`App.js: Initializing lesson: ${lessonId}`);

   // Construct the expected data variable name (e.g., 'week1Data')
   const lessonDataVarName = `${lessonId}Data`;
   const lessonData = window[lessonDataVarName];

   if (typeof lessonData === 'undefined') {
       console.error(`App.js: Lesson data object "${lessonDataVarName}" is not defined or lessonData file wasn't loaded correctly.`);
       displayErrorMessage(`Could not load data for ${lessonId}.`);
       return;
   }

   // --- Initialize Activity Modules ---
   initInteractiveChart(lessonData);
   initMatchingGame(lessonData);
   initSoundComparison(lessonData);
   initWritingPractice(lessonData);
   // Add calls to initialize other activity types here later

   console.log(`App.js: ${lessonId} initialization routines complete.`);
}

/**
* Initializes the Interactive Chart module if data and elements exist.
* @param {object} lessonData - The data object for the current lesson.
*/
function initInteractiveChart(lessonData) {
   const sectionElement = document.getElementById('activity-chart');
   if (!sectionElement) return; // Exit if section doesn't exist

   if (lessonData.alphabetChart && typeof InteractiveChart !== 'undefined') {
       const targetElement = document.getElementById(lessonData.alphabetChart.targetElementId);
       if (targetElement) {
           InteractiveChart.init(
               lessonData.alphabetChart.targetElementId,
               lessonData.alphabetChart.letters
           );
       } else {
           console.warn(`App.js: Target element "#${lessonData.alphabetChart.targetElementId}" for InteractiveChart not found.`);
           sectionElement.innerHTML = '<p>Error: Chart display area not found.</p>';
       }
   } else {
        console.log("App.js: InteractiveChart module or alphabetChart data not defined for this lesson.");
        sectionElement.innerHTML = '<p>Alphabet chart activity not available for this lesson.</p>';
   }
}

/**
* Initializes the Sound Comparison buttons if data and elements exist.
* @param {object} lessonData - The data object for the current lesson.
*/
function initSoundComparison(lessonData) {
   const sectionElement = document.getElementById('activity-comparison');
   if (!sectionElement) return;

   if (lessonData.soundComparison && typeof AudioPlayer !== 'undefined') {
       const soundButtonsContainer = document.getElementById('sound-comparison-audio-controls');
       const soundTable = document.getElementById('sound-comparison-table'); // Get table too

       // Option 1: If buttons are dynamically created
       if (soundButtonsContainer) {
           soundButtonsContainer.innerHTML = ''; // Clear placeholder
           if (lessonData.soundComparison.sounds && lessonData.soundComparison.sounds.length > 0) {
               lessonData.soundComparison.sounds.forEach(sound => {
                   const button = document.createElement('button');
                   button.className = 'button-like sound-compare-button';
                   button.textContent = sound.label || `Hear ${sound.pairId}`;
                   button.addEventListener('click', () => playComparisonSound(sound));
                   soundButtonsContainer.appendChild(button);
               });
           } else {
                soundButtonsContainer.innerHTML = '<p>No comparison sounds defined for this lesson.</p>';
           }
       }
       // Option 2: If buttons are static in the table (like in week1.html example)
       else if (soundTable) {
            const staticButtons = soundTable.querySelectorAll('.sound-compare-button[data-sound-pair]');
            if (staticButtons.length > 0) {
                staticButtons.forEach(button => {
                    const pairId = button.getAttribute('data-sound-pair');
                    const soundData = lessonData.soundComparison.sounds.find(s => s.pairId === pairId);
                    if (soundData) {
                        button.addEventListener('click', () => playComparisonSound(soundData));
                    } else {
                        button.disabled = true;
                        button.textContent = 'N/A';
                    }
                });
            } else if (!soundButtonsContainer) { // Only warn if neither dynamic nor static found
                 console.warn(`App.js: Could not find dynamic container (#${lessonData.soundComparison.targetElementId}) or static buttons for Sound Comparison.`);
            }
       }

   } else {
      console.log("App.js: Sound Comparison data or AudioPlayer module not defined for this lesson.");
      sectionElement.innerHTML = '<p>Sound comparison activity not available for this lesson.</p>';
   }
}

/** Helper function for playing comparison sounds */

       // Logic for Hebrew sound (e.g., play after delay, require second click, etc.)
       if (soundData.hebrewAudio) {
           console.log(`App.js: Comparison sound available: ${soundData.hebrewAudio}`);
           // Example: Play after 1 second
           // setTimeout(() => AudioPlayer.play(soundData.hebrewAudio), 1000);
       }
   } else {
       console.warn(`App.js: No English audio defined for sound pair ${soundData.pairId}`);
   }
}


/**
* Initializes the Writing Practice module if data and elements exist.
* @param {object} lessonData - The data object for the current lesson.
*/
function initWritingPractice(lessonData) {
   const sectionElement = document.getElementById('activity-writing');
   if (!sectionElement) return;

   if (lessonData.writingPractice && typeof WritingPractice !== 'undefined') {
       const canvasElement = document.getElementById(lessonData.writingPractice.targetCanvasId);
       const selectorContainer = document.getElementById('writing-letter-selector');
       const clearButton = document.getElementById('clear-canvas-button');

       if (canvasElement && selectorContainer && clearButton) {
          // Dynamically create letter selector buttons
         selectorContainer.innerHTML = ''; // Clear placeholder
         if (lessonData.writingPractice.guideLetters && Object.keys(lessonData.writingPractice.guideLetters).length > 0) {
             Object.keys(lessonData.writingPractice.guideLetters).sort().forEach(letter => {
                 const button = document.createElement('button');
                 button.className = 'button-like letter-select-button';
                 const lower = letter.toLowerCase();
                 button.textContent = `${letter} ${lower}`;
                 button.setAttribute('data-trace-letter', letter);
                 selectorContainer.appendChild(button);
             });

             // Initialize the WritingPractice module
             WritingPractice.init(
                 lessonData.writingPractice.targetCanvasId,
                 'clear-canvas-button',
                 '#writing-letter-selector button[data-trace-letter]', // Selector for dynamic buttons
                 lessonData.writingPractice.guideLetters
             );
         } else {
              selectorContainer.innerHTML = '<p>No letters available for writing practice.</p>';
              canvasElement.style.display = 'none';
              clearButton.style.display = 'none';
         }

// js/modules/audioPlayer.js - Handles audio playback

const AudioPlayer = {
   currentAudio: null,

   /**
    * Plays an audio file.
    * @param {string} audioSrc - Path to the audio file.
    * @param {boolean} [stopPrevious=true] - Whether to stop any currently playing audio.
    */
   play: function(audioSrc, stopPrevious = true) {
       if (!audioSrc || audioSrc.includes('placeholder')) { // Avoid playing placeholders
           console.warn(`AudioPlayer: Invalid or placeholder audio source: ${audioSrc}`);
           return;
       }

       if (stopPrevious && this.currentAudio) {
           this.stop();
       }

       try {
           this.currentAudio = new Audio(audioSrc);
           this.currentAudio.play()
               .then(() => {
                   console.log(`AudioPlayer: Playing ${audioSrc}`);
               })
               .catch(error => {
                   console.error(`AudioPlayer: Error playing ${audioSrc}:`, error);
                   this.currentAudio = null;
               });

           this.currentAudio.onended = () => {
               this.currentAudio = null;
           };
            this.currentAudio.onerror = (e) => {
                console.error(`AudioPlayer: Error loading audio source ${audioSrc}`, e);
                this.currentAudio = null;
            };

       } catch (error) {
           console.error(`AudioPlayer: Could not create Audio object for ${audioSrc}:`, error);
           this.currentAudio = null;
       }
   },

   /** Stops the currently playing audio, if any. */
   stop: function() {
       if (this.currentAudio) {
           try {
               this.currentAudio.pause();
               this.currentAudio.currentTime = 0;
               console.log("AudioPlayer: Stopped previous audio.");
           } catch (error) {
               console.error("AudioPlayer: Error stopping audio:", error);
           } finally {
               this.currentAudio = null;
           }
       }
   }
};

const MatchingGame = {
   pairs: [],
   gameType: 'click-pair',
   container: null,
   selectedItem: null,
   matchedPairs: 0,
   totalPairs: 0,
   lockBoard: false, // Prevents clicking during mismatch timeout

   /** Initializes a matching game. */
   init: function(containerId, pairsData, gameType = 'click-pair') {
       this.container = document.getElementById(containerId);
       if (!this.container) {
           console.error(`MatchingGame: Container element with ID "${containerId}" not found.`);
           return;
       }
       if (!pairsData || pairsData.length === 0) {
            console.warn(`MatchingGame: No pairs data provided for container "${containerId}".`);
            this.container.innerHTML = '<p>No matching items available for this lesson.</p>';
            return;
       }

       console.log(`MatchingGame: Initializing game type "${gameType}" in "#${containerId}".`);
       this.container.innerHTML = ''; // Clear placeholder

       this.pairs = pairsData;
       this.gameType = gameType;
       this.selectedItem = null;
       this.matchedPairs = 0;
       this.totalPairs = pairsData.length;
       this.lockBoard = false;

       const items = this.shuffleItems(pairsData);

       if (gameType === 'click-pair') {
           this.createClickPairGame(items);
       } else {
           console.warn(`MatchingGame: Game type "${gameType}" not yet implemented.`);
           this.container.innerHTML = `<p>Game type "${gameType}" coming soon!</p>`;
       }
   },

   /** Shuffles items for random display. */
   shuffleItems: function(pairsData) {
       let items = [];
       pairsData.forEach(pair => {
           // Ensure unique IDs even if content is the same (e.g., word-picture match)
           const id1 = pair.id1 || `item-${pair.item1}-${Math.random().toString(16).slice(2)}`;
           const id2 = pair.id2 || `item-${pair.item2}-${Math.random().toString(16).slice(2)}`;
           items.push({ id: id1, content: pair.item1, pairId: id2 });
           items.push({ id: id2, content: pair.item2, pairId: id1 });
       });

       for (let i = items.length - 1; i > 0; i--) {
           const j = Math.floor(Math.random() * (i + 1));
           [items[i], items[j]] = [items[j], items[i]];
       }
       return items;
   },

   /** Creates the HTML elements for a 'click-pair' game. */
   createClickPairGame: function(items) {
       const gameGrid = document.createElement('div');
       gameGrid.className = 'matching-game-grid';

       items.forEach(item => {
           const itemElement = document.createElement('div');
           // Use button-like class but maybe override specific styles in CSS if needed
           itemElement.className = 'matching-item button-like';
           itemElement.innerHTML = item.content; // Use innerHTML in case content has HTML (e.g., images)
           itemElement.setAttribute('data-id', item.id);
           itemElement.setAttribute('data-pair-id', item.pairId);

           itemElement.addEventListener('click', (e) => this.handleItemClick(e.currentTarget)); // Use currentTarget
           gameGrid.appendChild(itemElement);
       });

       this.container.appendChild(gameGrid);
   },

   /** Handles clicks on items in the 'click-pair' game. */
   handleItemClick: function(clickedElement) {
       if (this.lockBoard || clickedElement.classList.contains('matched') || clickedElement === this.selectedItem) {
           return; // Ignore click if board locked, already matched, or same item clicked twice
       }

       clickedElement.classList.add('selected');

       if (!this.selectedItem) {
           this.selectedItem = clickedElement;
       } else {
           // Second item selected - check for match
           this.lockBoard = true; // Lock board while checking
           const firstItemPairId = this.selectedItem.getAttribute('data-pair-id');
           const secondItemId = clickedElement.getAttribute('data-id');

           if (secondItemId === firstItemPairId) {
               // Match found
               console.log("MatchingGame: Match found!");
               this.selectedItem.classList.replace('selected', 'matched');
               clickedElement.classList.replace('selected', 'matched');
               if (typeof AudioPlayer !== 'undefined') AudioPlayer.play('../assets/audio/feedback/correct.mp3'); // Example sound path
               this.matchedPairs++;
               this.resetSelection();
               this.checkCompletion();
               this.lockBoard = false; // Unlock board
           } else {
               // No match
               console.log("MatchingGame: No match.");
                if (typeof AudioPlayer !== 'undefined') AudioPlayer.play('../assets/audio/feedback/incorrect.mp3'); // Example sound path
               // Briefly show selection, then hide
               setTimeout(() => {
                   this.selectedItem.classList.remove('selected');
                   clickedElement.classList.remove('selected');
                   this.resetSelection();
                   this.lockBoard = false; // Unlock board only after timeout
               }, 800); // Adjust delay
           }
       }
   },

   /** Resets the selected item state. */
   resetSelection: function() {
        this.selectedItem = null;
   },

   /** Checks if all pairs have been matched. */
   checkCompletion: function() {
       if (this.matchedPairs === this.totalPairs) {
           console.log("MatchingGame: All pairs matched! Game complete.");
           // Add completion message inside the container
           const completionMessage = document.createElement('p');
           completionMessage.textContent = 'Well Done! All matched!';
           completionMessage.style.color = 'green';
           completionMessage.style.fontWeight = 'bold';
           completionMessage.style.textAlign = 'center';
           completionMessage.style.marginTop = '15px';
           this.container.appendChild(completionMessage);
            if (typeof AudioPlayer !== 'undefined') AudioPlayer.play('../assets/audio/feedback/complete.mp3'); // Example sound path
       }
   }
};

// js/modules/writingPractice.js - Logic for canvas-based writing/tracing practice

const WritingPractice = {
   canvas: null,
   ctx: null,
   drawing: false,
   lastX: 0,
   lastY: 0,
   guideImage: null,
   currentLetter: null,
   guideLetterData: {}, // Store guide data locally

   /** Initializes the writing practice canvas. */
   init: function(canvasId, clearButtonId, letterButtonSelector, guideLetterData) {
       this.canvas = document.getElementById(canvasId);
       const clearButton = document.getElementById(clearButtonId);
       const letterButtons = document.querySelectorAll(letterButtonSelector); // Select buttons dynamically

       if (!this.canvas) { console.error(`WritingPractice: Canvas "#${canvasId}" not found.`); return; }
       if (!clearButton) { console.warn(`WritingPractice: Clear button "#${clearButtonId}" not found.`); }
       if (!letterButtons.length) { console.warn(`WritingPractice: No letter buttons found with selector "${letterButtonSelector}".`); }
       if (!guideLetterData || Object.keys(guideLetterData).length === 0) { console.warn(`WritingPractice: No guide letter data provided.`); }

       this.ctx = this.canvas.getContext('2d');
       this.guideLetterData = guideLetterData || {}; // Store guide data

       this.setupCanvasStyles();
       this.addEventListeners();

       if (clearButton) {
           clearButton.addEventListener('click', () => this.clearCanvas());
       }

       // Add listeners for dynamically created letter selection buttons
       letterButtons.forEach(button => {
           button.addEventListener('click', () => {
               const letter = button.getAttribute('data-trace-letter');
               this.loadLetterGuide(letter);
           });
       });

       console.log(`WritingPractice: Initialized canvas "#${canvasId}".`);
       this.clearCanvas(); // Start clear
   },

   /** Sets initial drawing styles. */
   setupCanvasStyles: function() {
       this.ctx.strokeStyle = '#333333'; // Darker ink
       this.ctx.lineWidth = 4; // Slightly thicker
       this.ctx.lineJoin = 'round';
       this.ctx.lineCap = 'round';
   },

   /** Adds mouse and touch event listeners for drawing. */
   addEventListeners: function() {
       const canvas = this.canvas; // Cache canvas element

       // Mouse Events
       canvas.addEventListener('mousedown', (e) => this.startDrawing(this.getCoords(e)));
       canvas.addEventListener('mousemove', (e) => this.draw(this.getCoords(e)));
       canvas.addEventListener('mouseup', () => this.stopDrawing());
       canvas.addEventListener('mouseout', () => this.stopDrawing());

       // Touch Events
       canvas.addEventListener('touchstart', (e) => {
            if (e.touches.length === 1) { // Handle single touch
                this.startDrawing(this.getCoords(e.touches[0]));
            }
        }, { passive: true }); // Improve scroll performance

       canvas.addEventListener('touchmove', (e) => {
            if (e.touches.length === 1) {
                e.preventDefault(); // Prevent page scroll ONLY when drawing
                this.draw(this.getCoords(e.touches[0]));
            }
        }, { passive: false }); // Need false to preventDefault

       canvas.addEventListener('touchend', () => this.stopDrawing());
       canvas.addEventListener('touchcancel', () => this.stopDrawing()); // Handle interruption
   },

   /** Gets coordinates relative to the canvas. */
   getCoords: function(e) {
       const rect = this.canvas.getBoundingClientRect();
       // Adjust for potential CSS scaling if necessary (more complex)
       // const scaleX = this.canvas.width / rect.width;
       // const scaleY = this.canvas.height / rect.height;
       return {
           // x: (e.clientX - rect.left) * scaleX,
           // y: (e.clientY - rect.top) * scaleY
            x: e.clientX - rect.left,
            y: e.clientY - rect.top
       };
   },

   /** Starts the drawing process. */
   startDrawing: function(coords) {
       this.drawing = true;
       // Begin path immediately to draw starting dot/line segment correctly
       this.ctx.beginPath();
       this.ctx.moveTo(coords.x, coords.y); // Move to start point
       [this.lastX, this.lastY] = [coords.x, coords.y];
   },

   /** Draws a line from the last point to the current point. */
   draw: function(coords) {
       if (!this.drawing) return;
       this.ctx.lineTo(coords.x, coords.y); // Add line segment to current path
       this.ctx.stroke(); // Draw the path defined so far
       // Update last coordinates for the next segment
       [this.lastX, this.lastY] = [coords.x, coords.y];
   },

   /** Stops the drawing process. */
   stopDrawing: function() {
       if (this.drawing) {
            this.ctx.closePath(); // Close the current path if needed
            this.drawing = false;
       }
   },

   /** Clears the user's drawing, keeping the guide image. */
   clearCanvas: function() {
       // Store current guide state
       const currentGuideImage = this.guideImage;
       const currentLetter = this.currentLetter;

       // Clear the entire canvas
       this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);

       // If a guide was loaded and is ready, redraw it
       if (currentGuideImage && currentGuideImage.complete) {
           try {
               this.ctx.drawImage(currentGuideImage, 0, 0, this.canvas.width, this.canvas.height);
               console.log(`WritingPractice: Redrew guide for "${currentLetter}".`);
           } catch (e) {
               console.error("WritingPractice: Error redrawing guide image.", e);
               this.guideImage = null; // Reset if error drawing
               this.currentLetter = null;
           }
       } else if (currentLetter) {
           // If guide wasn't ready, try loading it again (might happen on fast clears)
           console.log(`WritingPractice: Guide for "${currentLetter}" wasn't ready, attempting reload on clear.`);
           this.loadLetterGuide(currentLetter); // This will clear again once loaded
       }
       console.log("WritingPractice: User drawing cleared.");
   },

   /** Loads and displays the tracing guide image. */
   loadLetterGuide: function(letter) {
       if (!this.guideLetterData || !this.guideLetterData[letter]) {
           console.warn(`WritingPractice: No guide data for letter "${letter}". Clearing canvas.`);
           this.guideImage = null;
           this.currentLetter = null;
           this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height); // Clear canvas fully
           return;
       }

       this.currentLetter = letter;
       // Simplified: Assume one guide image per letter (e.g., combined upper/lower)
       // Prioritize 'upper' if both exist, adjust as needed.
       const imagePath = this.guideLetterData[letter].upper || this.guideLetterData[letter].lower || this.guideLetterData[letter].combined;

       if (!imagePath || imagePath.includes('placeholder')) {
           console.warn(`WritingPractice: Invalid or placeholder image path for letter "${letter}". Clearing canvas.`);
           this.guideImage = null;
           this.currentLetter = null;
           this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);
           return;
       }

       console.log(`WritingPractice: Loading guide for "${letter}" from ${imagePath}`);

       // Create a new Image object for the guide
       const img = new Image();
       img.onload = () => {
           console.log(`WritingPractice: Guide image for "${letter}" loaded successfully.`);
           this.guideImage = img; // Store the loaded image object
           this.clearCanvas(); // Clear previous drawing and draw the new guide background
       };
       img.onerror = () => {
           console.error(`WritingPractice: Failed to load guide image: ${imagePath}`);
           // Clear canvas and reset guide state if image fails to load
           if (this.currentLetter === letter) { // Avoid clearing if another letter was selected quickly
               this.guideImage = null;
               this.currentLetter = null;
               this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);
           }
       };
       img.src = imagePath; // Start loading the image
   }
};
